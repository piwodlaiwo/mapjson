Sketched Earth

The world as if drawn by hand in a field notebook. The same coast-following contours as Rings, but every line is redrawn as a loose, wobbly pen stroke — doubled and slightly offset the way you'd retrace a line on paper — in warm ink on a toned page. Big landmasses fill with nested contours; the ocean stays blank. One /v1/geo call, sketched.

How it's made

One mapjson call is rasterized to a land mask; a distance transform grows a field inward from every coast, and d3.contours pulls nested rings from it. Then each ring is sketched: its points are jittered by a pixel or two, smoothed through a Catmull-Rom curve, and stroked twice with a translucent nib so the two passes overlap like a hand retracing the line. No fills, no straight edges — just ink that wobbles. Re-sketch for a fresh hand.

API Calls

https://api.mapjson.com/v1/geo?layer=countries&filter=world&detail=high&format=geojson

  <script src="https://cdn.jsdelivr.net/npm/d3@7/dist/d3.min.js"></script>
  <script>
  const W = 960, H = 560;
  const svg = d3.select("#map").append("svg").attr("viewBox", `0 0 ${W} ${H}`);
  const off = document.createElement("canvas"); off.width = W; off.height = H;
  const octx = off.getContext("2d");

  const GW = 240, GH = 140, cw = W / GW, ch = H / GH;
  let D = null;
  function buildField(world) {
    const proj = d3.geoEqualEarth().rotate([-11, 0]).fitExtent([[26, 22], [W - 26, H - 22]], { type: "Sphere" });
    const p = d3.geoPath(proj, octx);
    octx.fillStyle = "#000"; octx.fillRect(0, 0, W, H);
    octx.beginPath(); world.features.forEach((f) => p(f)); octx.fillStyle = "#fff"; octx.fill();
    const m = octx.getImageData(0, 0, W, H).data;
    const B = new Uint8Array(GW * GH);
    for (let gy = 0; gy < GH; gy++) for (let gx = 0; gx < GW; gx++) {
      const x = ((gx + 0.5) * cw) | 0, y = ((gy + 0.5) * ch) | 0;
      B[gy * GW + gx] = m[(y * W + x) * 4] > 128 ? 1 : 0;
    }
    D = new Float32Array(GW * GH);
    for (let i = 0; i < D.length; i++) D[i] = B[i] ? 1e5 : 0;
    for (let gy = 0; gy < GH; gy++) for (let gx = 0; gx < GW; gx++) {
      const i = gy * GW + gx; if (!B[i]) continue;
      let v = D[i]; if (gx > 0) v = Math.min(v, D[i - 1] + 1); if (gy > 0) v = Math.min(v, D[i - GW] + 1); D[i] = v;
    }
    for (let gy = GH - 1; gy >= 0; gy--) for (let gx = GW - 1; gx >= 0; gx--) {
      const i = gy * GW + gx; if (!B[i]) continue;
      let v = D[i]; if (gx < GW - 1) v = Math.min(v, D[i + 1] + 1); if (gy < GH - 1) v = Math.min(v, D[i + GW] + 1); D[i] = v;
    }
  }

  const S = W / GW;                          // grid → canvas scale
  const line = d3.line().curve(d3.curveCatmullRomClosed.alpha(0.5));

  function paint() {
    svg.selectAll("*").remove();
    svg.append("rect").attr("width", W).attr("height", H).attr("fill", "#f3ecdd");
    const maxD = d3.max(D) || 1;
    const contours = d3.contours().size([GW, GH]).thresholds(d3.range(0.5 + Math.random() * 2, maxD, 2.2 + Math.random() * 0.6))(Array.from(D));
    const g = svg.append("g").attr("fill", "none").attr("stroke", "#2b271f").attr("stroke-linecap", "round").attr("stroke-linejoin", "round");

    contours.forEach((c, ci) => {
      const amp = ci === 0 ? 1.7 : 1.15, w = ci === 0 ? 1.35 : 0.85;
      c.coordinates.forEach((poly) => poly.forEach((ring) => {
        if (ring.length < 6) return;
        for (let pass = 0; pass < 2; pass++) {
          const pts = ring.map(([x, y]) => [x * S + (Math.random() * 2 - 1) * amp, y * S + (Math.random() * 2 - 1) * amp]);
          g.append("path").attr("d", line(pts)).attr("stroke-width", w).attr("stroke-opacity", 0.5);
        }
      }));
    });
  }

  d3.json("https://api.mapjson.com/v1/geo?layer=countries&filter=world&detail=high&format=geojson")
    .then((w) => { buildField(w); paint(); });
  document.getElementById("repaint").addEventListener("click", paint);
  </script>
