Op-Art Earth: Spectrum II

The flowing spectrum of Spectrum, but poured only into the land. Real coastlines this time — the rippling color bands are clipped to the continents and the sea goes dark, so it reads unmistakably as a world map lit up in rainbow. The oceans hold the shape; the color does the shimmer. One /v1/geo call.

How it's made

One mapjson call gives the country polygons. They're turned into an SVG clipPath — the exact silhouette of every landmass — and the flowing spectrum bands are drawn only inside it, over a near-black sea. A soft white coastline traces the edge so the continents stay crisp. Because the map shape is now the real geometry rather than an implied ripple, it's legible first and optical second. Re-run to shift the wave.

API Calls

https://api.mapjson.com/v1/geo?layer=countries&filter=world&detail=high&format=geojson

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <title>Op-Art Earth: Spectrum II — chromatic world map</title>
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>
    body { margin: 0; background: #fafafa; font-family: system-ui, sans-serif; }
    .wrap { max-width: 980px; margin: 2rem auto; padding: 0 1rem; }
    #map { background: #0e1116; border-radius: 4px; overflow: hidden; }
    #map svg { width: 100%; display: block; }
    button { margin-top: 0.7rem; font: 12px ui-monospace, monospace; padding: 6px 12px; background: #fff; border: 1px solid #ddd; border-radius: 4px; cursor: pointer; }
  </style>
</head>
<body>
  <div class="wrap">
    <div id="map"></div>
    <button id="repaint">↻ re-run</button>
  </div>
  <script src="https://cdn.jsdelivr.net/npm/d3@7/dist/d3.min.js"></script>
  <script>
  const W = 960, H = 560;
  const svg = d3.select("#map").append("svg").attr("viewBox", `0 0 ${W} ${H}`);
  const PAL = ["#d1394f", "#e17b3a", "#e8c33a", "#5fa85a", "#2f9c95", "#3f6fb0", "#7a4fa0"];
  let feats = null, path = null;

  function paint() {
    svg.selectAll("*").remove();
    svg.append("rect").attr("width", W).attr("height", H).attr("fill", "#0e1116");
    if (!feats) return;
    const defs = svg.append("defs");
    defs.append("clipPath").attr("id", "land").selectAll("path").data(feats).join("path").attr("d", path);

    const g = svg.append("g").attr("clip-path", "url(#land)");
    const bh = H / 30, k = (2 * Math.PI) / (170 + Math.random() * 90), ph = Math.random() * 6.283, A = bh * 0.8;
    const by = (j, x) => j * bh + A * Math.sin(x * k + j * 0.5 + ph);
    const n = Math.ceil(H / bh) + 1;
    for (let j = 0; j < n; j++) {
      let d = "";
      for (let x = 0; x <= W; x += 5) d += (x === 0 ? "M" : "L") + x + "," + by(j, x).toFixed(1);
      for (let x = W; x >= 0; x -= 5) d += "L" + x + "," + by(j + 1, x).toFixed(1);
      g.append("path").attr("d", d + "Z").attr("fill", PAL[j % PAL.length]);
    }
    // crisp coastline over the dark sea
    svg.append("g").selectAll("path").data(feats).join("path")
      .attr("d", path).attr("fill", "none").attr("stroke", "rgba(255,255,255,0.28)").attr("stroke-width", 0.6);
  }

  d3.json("https://api.mapjson.com/v1/geo?layer=countries&filter=world&detail=high&format=geojson").then((w) => {
    feats = w.features;
    path = d3.geoPath(d3.geoEqualEarth().rotate([-11, 0]).fitExtent([[24, 20], [W - 24, H - 20]], { type: "Sphere" }));
    paint();
  });
  document.getElementById("repaint").addEventListener("click", paint);
  </script>
</body>
</html>
